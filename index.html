    <script>

        // ローカルストレージを使った簡易同期（クラウド連携のベース）

        let players = [

            { name: "プレイヤー1", score: 25000 },

            { name: "プレイヤー2", score: 25000 },

            { name: "プレイヤー3", score: 25000 },

            { name: "プレイヤー4", score: 25000 }

        ];


        function updateDisplay() {

            const table = document.getElementById('ranking-table');

            // スコア順にソート

            players.sort((a, b) => b.score - a.score);


            let html = '';

            players.forEach((p, index) => {

                html += `<tr>

                    <td>${index + 1}位</td>

                    <td>${p.name}</td>

                    <td>${p.score.toLocaleString()} 点</td>

                </tr>`;

            });

            table.innerHTML = html;

        }


        document.getElementById('score-form').addEventListener('submit', function(e) {

            e.preventDefault();

            const winner = document.getElementById('winner').value;

            const points = parseInt(document.getElementById('points').value) || 0;


            // 簡単な加算処理（実際はここに細かいロン・ツモのやり取りロジックが入る）

            players.forEach(p => {

                if (p.name === winner) {

                    p.score += points;

                } else {

                    p.score -= Math.floor(points / 3); // 簡易的な減点

                }

            });


            updateDisplay();

            alert(winner + ' の勝利を反映しました！');

            document.getElementById('points').value = '';

        });


        function resetData() {

            if(confirm('スコアを初期化しますか？')) {

                players.forEach(p => p.score = 25000);

                updateDisplay();

            }

        }


        // 初回ロード時

        updateDisplay();

    </script>
